@pytest.fixture
def window(tmp_path,monkeypatch):
    app=QApplication.instance() or QApplication([])
    monkeypatch.setattr(QMessageBox,'warning',lambda *a,**kw:pytest.fail(str(a[2])))
    w=Window();w.show()
    source=make_video(tmp_path/'video.mp4',duration=4,fps=4)
    w.open_source(source.path)
    spin(app,lambda:w.controller.state==State.SOURCE_READY)
    yield app,w
    w.close();spin(app,lambda:not w.isVisible())


def review(app,w):
    w.analyze();spin(app,lambda:w.controller.state==State.REVIEW_READY)


def test_home_dialog_contains_analysis_settings_only(window,monkeypatch):
    app,w=window
    assert w.advanced_button.text()=='Analysis settings…'
    def inspect(dialog):
        assert dialog.windowTitle()=='Analysis settings'
        assert not dialog.findChildren(QCheckBox)
        combos=dialog.findChildren(QComboBox)
        assert len(combos)==1 and combos[0].count()==2
        combos[0].setCurrentIndex(1)
        return QDialog.DialogCode.Rejected
    monkeypatch.setattr(QDialog,'exec',inspect)
    settings=w.settings;task=w._sample_preparation
    w.open_advanced()
    assert w.settings==settings and w._sample_preparation is task
    assert not w.json_enabled


@pytest.mark.parametrize('accept',[False,True])
def test_save_options_preserve_edited_project(window,monkeypatch,accept):
    app,w=window;review(app,w)
    assert w.save_options_button.isVisible() and not w.advanced_button.isVisible()
    project=w.controller.project;w.delete_selected_pages();w.undo_edit()
    pages=tuple(project.pages);revision=project.revision;history=tuple(w.history.undo_stack)
    task=w._sample_preparation;settings=w.settings
    def inspect(dialog):
        assert dialog.windowTitle()=='Save options'
        boxes=dialog.findChildren(QCheckBox);combos=dialog.findChildren(QComboBox)
        assert len(boxes)==1 and len(combos)==1
        assert combos[0].itemText(0)=='Simple slide times'
        assert not combos[0].isEnabled()
        boxes[0].setChecked(True);assert combos[0].isEnabled()
        combos[0].setCurrentIndex(1)
        return QDialog.DialogCode.Accepted if accept else QDialog.DialogCode.Rejected
    monkeypatch.setattr(QDialog,'exec',inspect)
    w.open_save_options()
    assert w.controller.project is project and w._sample_preparation is task
    assert tuple(project.pages)==pages and project.revision==revision
    assert tuple(w.history.undo_stack)==history and w.settings==settings
    assert w.controller.state==State.REVIEW_READY and w.export_button.isEnabled()
    assert w.json_enabled==accept and w.json_format==('detailed' if accept else 'simple')
    assert w.export_button.text()==('Save PDF + JSON' if accept else 'Save PDF')


def test_home_cancel_preserves_current_work(window,monkeypatch):
    app,w=window;review(app,w)
    project=w.controller.project;task=w._sample_preparation
    before=w._current_image;history=tuple(w.history.undo_stack)
    monkeypatch.setattr(QMessageBox,'question',lambda *a,**kw:QMessageBox.StandardButton.No)
    w.back_to_preparation.click()
    assert w.stage=='review' and w.controller.project is project
    assert w._sample_preparation is task and not task.token.cancelled
    assert tuple(w.history.undo_stack)==history and w._current_image is before
    assert task.result.directory.exists()


def test_home_confirmation_clears_work_and_cache(window,monkeypatch,tmp_path):
    app,w=window;review(app,w)
    task=w._sample_preparation;directory=task.result.directory
    old_key=w._image_key(w.current_index);old_image=w._current_image
    questions=[]
    def confirm(*a,**kw):questions.append(a);return QMessageBox.StandardButton.Yes
    monkeypatch.setattr(QMessageBox,'question',confirm)
    w.back_to_preparation.click()
    assert questions and 'saved' in questions[0][2].lower()
    assert w.stage=='empty' and w.controller.state==State.EMPTY
    assert w.controller.project is None and w.controller.source is None
    assert w._first_frame is None and w.player is None
    assert w.page_model.rowCount()==0 and not w.history.undo_stack and not w.history.redo_stack
    assert w.open_button.isEnabled() and not w.analyze_button.isEnabled()
    assert not w.save_options_button.isVisible() and not w.timeline.isVisible()
    w._image_ready(old_key,old_image,None)
    assert w._current_image is None
    spin(app,lambda:task.done.is_set())
    assert not directory.exists()
    w.open_source(make_video(tmp_path/'next.mp4',duration=2,fps=4).path)
    spin(app,lambda:w.controller.state==State.SOURCE_READY)
    w.analyze();spin(app,lambda:w.controller.state==State.REVIEW_READY)
    assert len(w.controller.project.samples)==1


@pytest.mark.parametrize('format',['simple','detailed'])
def test_save_options_export_json_and_home_keeps_output(window,monkeypatch,tmp_path,format):
    app,w=window;review(app,w)
    def configure(dialog):
        dialog.findChild(QCheckBox).setChecked(True)
        dialog.findChild(QComboBox).setCurrentIndex(0 if format=='simple' else 1)
        return QDialog.DialogCode.Accepted
    monkeypatch.setattr(QDialog,'exec',configure)
    w.open_save_options()
    target=tmp_path/'slides.pdf'
    monkeypatch.setattr(QFileDialog,'getSaveFileName',lambda *a,**kw:(str(target),'PDF'))
    w.export_pending();spin(app,lambda:w.stage=='complete',timeout=30)
    data=json.loads(target.with_suffix('.json').read_text())
    assert ('schema_version' in data)==(format=='detailed')
    assert w.new_video_button.text()=='Start over'
    monkeypatch.setattr(QMessageBox,'question',lambda *a,**kw:QMessageBox.StandardButton.Yes)
    w.new_video_button.click()
    assert w.stage=='empty' and w.controller.project is None
    assert target.is_file() and target.with_suffix('.json').is_file()
    assert w._last_export is None
