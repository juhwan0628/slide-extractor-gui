cmake_minimum_required(VERSION 3.16)
project(slide_keyframe_decode LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavfilter libavutil)
add_executable(slide-keyframe-decode keyframe_decode.cpp)
target_compile_features(slide-keyframe-decode PRIVATE cxx_std_17)
target_link_libraries(slide-keyframe-decode PRIVATE PkgConfig::FFMPEG)
if(NOT MSVC)
  target_compile_options(slide-keyframe-decode PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  target_link_libraries(slide-keyframe-decode PRIVATE shell32)
endif()
